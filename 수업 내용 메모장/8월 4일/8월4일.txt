중첩문 만드는 방법1
rows = []

for user in data:
    row = {
        "name": user["name"],
        "email": user["email"],
        "city": user["address"]["city"],       # city는 address 안에 중첩되어 있음
        "company": user["company"]["name"],    # company도 company 안에 중첩되어 있음
    }
    rows.append(row)

print(rows)
